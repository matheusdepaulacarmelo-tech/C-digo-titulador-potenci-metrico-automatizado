String nomeEstado() {

  if (titulacaoPausada)
    return "PAUSADO";

  switch (estadoAtual) {

    case PARADO:
      return "PARADO";

    case DOSANDO:
      return "DOSANDO";

    case HOMOGENEIZANDO:
      return "HOMOGENEIZANDO";

    case ESTABILIZANDO:
      return "ESTABILIZANDO";

    case ADQUIRINDO:
      return "ADQUIRINDO";

    case FINALIZADO:
      return "FINALIZADO";

    case INTERROMPIDO:
      return "INTERROMPIDO";
  }

  return "---";
}

// ======================================================
// CONTROLE DO DRIVER
// ======================================================

void habilitarMotor() {

  // ENABLE do A4988 é ativo em nível baixo

  digitalWrite(
    PIN_ENABLE,
    LOW
  );
}

void desabilitarMotor() {

  digitalWrite(
    PIN_ENABLE,
    HIGH
  );
}

// ======================================================
// LEITURA DO POTENCIAL
// ======================================================

float lerPotencialMv() {

  const int numeroLeituras = 10;

  float soma = 0;

  for (int i = 0;
       i < numeroLeituras;
       i++) {

    int16_t leitura =
      ads.readADC_Differential_0_1();

    float tensao =
      ads.computeVolts(leitura);

    soma +=
      tensao * 1000.0;

    delay(5);
  }

  float potencial =
    soma / numeroLeituras;

  potencial =
    potencial *
    ganhoPotencial +
    offsetPotencialMv;

  return potencial;
}

// ======================================================
// LEITURA DA TEMPERATURA
// ======================================================

float lerTemperatura() {

  sensorTemperatura.requestTemperatures();

  return
    sensorTemperatura.getTempCByIndex(0);
}

// ======================================================
// PAUSA
// ======================================================

void aguardarEnquantoPausado() {

  while (
    titulacaoPausada &&
    titulacaoAtiva
  ) {

    desabilitarMotor();

    server.handleClient();

    delay(10);
  }
}

// ======================================================
// ESPERA SEM BLOQUEAR A INTERFACE
// ======================================================

bool esperarResponsivo(
  unsigned long tempoMs
) {

  unsigned long tempoDecorrido = 0;

  unsigned long anterior =
    millis();

  while (
    tempoDecorrido < tempoMs &&
    titulacaoAtiva
  ) {

    server.handleClient();

    if (titulacaoPausada) {

      aguardarEnquantoPausado();

      anterior =
        millis();

      continue;
    }

    unsigned long agora =
      millis();

    tempoDecorrido +=
      agora - anterior;

    anterior =
      agora;

    delay(2);
  }

  return titulacaoAtiva;
}

// ======================================================
// DOSAGEM
// ======================================================

float dosarVolume(
  float volumeDesejado
) {

  estadoAtual = DOSANDO;

  long passosDesejados =
    lround(
      volumeDesejado *
      passosPorMl
    );

  long passosExecutados = 0;

  digitalWrite(
    PIN_DIR,
    sentidoDosagem
  );

  habilitarMotor();

  for (
    long i = 0;
    i < passosDesejados;
    i++
  ) {

    if (!titulacaoAtiva)
      break;

    if (titulacaoPausada) {

      desabilitarMotor();

      aguardarEnquantoPausado();

      if (!titulacaoAtiva)
        break;

      habilitarMotor();
    }

    digitalWrite(
      PIN_STEP,
      HIGH
    );

    delayMicroseconds(
      tempoPulsoUs
    );

    digitalWrite(
      PIN_STEP,
      LOW
    );

    delayMicroseconds(
      tempoPulsoUs
    );

    passosExecutados++;

    // Mantém a interface responsiva
    if (
      passosExecutados % 25 == 0
    ) {

      server.handleClient();
    }
  }

  desabilitarMotor();

  return
    passosExecutados /
    passosPorMl;
}

// ======================================================
// ESTABILIZAÇÃO DO POTENCIAL
// ======================================================

bool aguardarEstabilizacao() {

  estadoAtual =
    ESTABILIZANDO;

  float potencialAnterior =
    lerPotencialMv();

  int leiturasEstaveis = 0;

  unsigned long tempoAtivo = 0;

  unsigned long ultimoInstante =
    millis();

  unsigned long ultimaLeitura =
    millis();

  while (
    titulacaoAtiva &&
    tempoAtivo <
    tempoMaxEstabilizacaoMs
  ) {

    server.handleClient();

    if (titulacaoPausada) {

      aguardarEnquantoPausado();

      ultimoInstante =
        millis();

      ultimaLeitura =
        millis();

      continue;
    }

    unsigned long agora =
      millis();

    tempoAtivo +=
      agora - ultimoInstante;

    ultimoInstante =
      agora;

    if (
      agora - ultimaLeitura >=
      intervaloEstabilizacaoMs
    ) {

      ultimaLeitura =
        agora;

      float potencialAtual =
        lerPotencialMv();

      float diferenca =
        fabs(
          potencialAtual -
          potencialAnterior
        );

      if (
        diferenca <=
        limiteEstabilidadeMv
      ) {

        leiturasEstaveis++;
      }

      else {

        leiturasEstaveis = 0;
      }

      potencialAnterior =
        potencialAtual;

      if (
        tempoAtivo >=
        tempoMinEstabilizacaoMs &&
        leiturasEstaveis >=
        numeroLeiturasEstaveis
      ) {

        return true;
      }
    }

    delay(2);
  }

  /*
    Caso o limite de tempo seja atingido,
    a sequência continua com a aquisição.
  */

  return titulacaoAtiva;
}

// ======================================================
// REGISTRO DE UM PONTO
// ======================================================

void registrarPonto() {

  estadoAtual =
    ADQUIRINDO;

  ultimoPotencialMv =
    lerPotencialMv();

  ultimaTemperatura =
    lerTemperatura();

  float instante =
    (millis() - tempoInicio)
    / 1000.0;

  if (
    numeroPontos <
    MAX_PONTOS
  ) {

    volumes[numeroPontos] =
      volumeAcumulado;

    potenciais[numeroPontos] =
      ultimoPotencialMv;

    temperaturas[numeroPontos] =
      ultimaTemperatura;

    tempos[numeroPontos] =
      instante;

    numeroPontos++;
  }

  // Envio também pela porta serial em formato CSV

  Serial.print(
    volumeAcumulado,
    4
  );

  Serial.print(",");

  Serial.print(
    ultimoPotencialMv,
    3
  );

  Serial.print(",");

  Serial.print(
    ultimaTemperatura,
    2
  );

  Serial.print(",");

  Serial.println(
    instante,
    2
  );
}

// ======================================================
// INICIAR TITULAÇÃO
// ======================================================

void iniciarTitulacao() {

  numeroPontos = 0;

  volumeAcumulado = 0;

  titulacaoPausada = false;

  titulacaoAtiva = true;

  estadoAtual =
    ADQUIRINDO;

  tempoInicio =
    millis();

  /*
    Registra o primeiro ponto antes da
    adição de titulante, correspondente
    a V = 0 mL.
  */

  registrarPonto();
}

// ======================================================
// FINALIZAÇÃO
// ======================================================

void finalizarTitulacao() {

  titulacaoAtiva = false;

  titulacaoPausada = false;

  desabilitarMotor();

  estadoAtual =
    FINALIZADO;
}

// ======================================================
// INTERRUPÇÃO
// ======================================================

void interromperTitulacao() {

  titulacaoAtiva = false;

  titulacaoPausada = false;

  desabilitarMotor();

  estadoAtual =
    INTERROMPIDO;
}

// ======================================================
// CICLO PRINCIPAL DA TITULAÇÃO
// ======================================================

void executarCicloTitulacao() {

  if (
    !titulacaoAtiva ||
    titulacaoPausada
  )
    return;

  if (
    volumeAcumulado >=
    volumeFinalMl
  ) {

    finalizarTitulacao();

    return;
  }

  float volumeRestante =
    volumeFinalMl -
    volumeAcumulado;

  float proximoIncremento =
    incrementoMl;

  if (
    proximoIncremento >
    volumeRestante
  ) {

    proximoIncremento =
      volumeRestante;
  }

  // -------------------------
  // DOSAGEM
  // -------------------------

  float volumeReal =
    dosarVolume(
      proximoIncremento
    );

  volumeAcumulado +=
    volumeReal;

  if (!titulacaoAtiva)
    return;

  // -------------------------
  // HOMOGENEIZAÇÃO
  // -------------------------

  /*
    O agitador magnético pode permanecer
    ligado externamente durante o ensaio.
    Aqui é aguardado apenas o período
    definido para homogeneização.
  */

  estadoAtual =
    HOMOGENEIZANDO;

  if (
    !esperarResponsivo(
      tempoHomogeneizacaoMs
    )
  )
    return;

  // -------------------------
  // ESTABILIZAÇÃO
  // -------------------------

  if (
    !aguardarEstabilizacao()
  )
    return;

  // -------------------------
  // AQUISIÇÃO
  // -------------------------

  registrarPonto();

  // -------------------------
  // VERIFICAÇÃO FINAL
  // -------------------------

  if (
    volumeAcumulado >=
    volumeFinalMl
  ) {

    finalizarTitulacao();
  }
}

void configurarServidor() {

  // Página principal

  server.on(
    "/",
    HTTP_GET,
    []() {

      server.send(
        200,
        "text/html",
        paginaWeb
      );
    }
  );

  // ----------------------------------------------
  // INICIAR
  // ----------------------------------------------

  server.on(
    "/iniciar",
    HTTP_POST,
    []() {

      if (
        server.hasArg(
          "incremento"
        )
      ) {

        incrementoMl =
          server.arg(
            "incremento"
          ).toFloat();
      }

      if (
        server.hasArg(
          "volumeFinal"
        )
      ) {

        volumeFinalMl =
          server.arg(
            "volumeFinal"
          ).toFloat();
      }

      if (
        server.hasArg(
          "homogeneizacao"
        )
      ) {

        float segundos =
          server.arg(
            "homogeneizacao"
          ).toFloat();

        tempoHomogeneizacaoMs =
          (unsigned long)
          (segundos * 1000.0);
      }

      iniciarTitulacao();

      server.send(
        200,
        "text/plain",
        "OK"
      );
    }
  );

  // ----------------------------------------------
  // PAUSAR / CONTINUAR
  // ----------------------------------------------

  server.on(
    "/pausar",
    HTTP_POST,
    []() {

      if (titulacaoAtiva) {

        titulacaoPausada =
          !titulacaoPausada;

        if (
          titulacaoPausada
        ) {

          desabilitarMotor();
        }
      }

      server.send(
        200,
        "text/plain",
        "OK"
      );
    }
  );

  // ----------------------------------------------
  // PARAR
  // ----------------------------------------------

  server.on(
    "/parar",
    HTTP_POST,
    []() {

      interromperTitulacao();

      server.send(
        200,
        "text/plain",
        "OK"
      );
    }
  );

  // ----------------------------------------------
  // DADOS ATUAIS
  // ----------------------------------------------

  server.on(
    "/dados",
    HTTP_GET,
    []() {

      String json = "{";

      json +=
        "\"estado\":\"" +
        nomeEstado() +
        "\",";

      json +=
        "\"volume\":" +
        String(
          volumeAcumulado,
          4
        ) +
        ",";

      json +=
        "\"potencial\":" +
        String(
          ultimoPotencialMv,
          3
        ) +
        ",";

      json +=
        "\"temperatura\":" +
        String(
          ultimaTemperatura,
          2
        );

      json += "}";

      server.send(
        200,
        "application/json",
        json
      );
    }
  );

  // ----------------------------------------------
  // CURVA
  // ----------------------------------------------

  server.on(
    "/curva",
    HTTP_GET,
    []() {

      String json = "[";

      for (
        int i = 0;
        i < numeroPontos;
        i++
      ) {

        json += "{";

        json +=
          "\"v\":" +
          String(
            volumes[i],
            4
          );

        json += ",";

        json +=
          "\"e\":" +
          String(
            potenciais[i],
            3
          );

        json += "}";

        if (
          i <
          numeroPontos - 1
        ) {

          json += ",";
        }
      }

      json += "]";

      server.send(
        200,
        "application/json",
        json
      );
    }
  );

  // ----------------------------------------------
  // CSV
  // ----------------------------------------------

  server.on(
    "/csv",
    HTTP_GET,
    []() {

      String csv =
        "Volume_mL,"
        "Potencial_mV,"
        "Temperatura_C,"
        "Tempo_s\n";

      for (
        int i = 0;
        i < numeroPontos;
        i++
      ) {

        csv +=
          String(
            volumes[i],
            4
          );

        csv += ",";

        csv +=
          String(
            potenciais[i],
            3
          );

        csv += ",";

        csv +=
          String(
            temperaturas[i],
            2
          );

        csv += ",";

        csv +=
          String(
            tempos[i],
            2
          );

        csv += "\n";
      }

      server.sendHeader(
        "Content-Disposition",
        "attachment; "
        "filename=titulacao.csv"
      );

      server.send(
        200,
        "text/csv",
        csv
      );
    }
  );

  server.begin();
}

// ======================================================
// SETUP
// ======================================================

void setup() {

  Serial.begin(115200);

  // ----------------------------------------------
  // DRIVER
  // ----------------------------------------------

  pinMode(
    PIN_ENABLE,
    OUTPUT
  );

  pinMode(
    PIN_STEP,
    OUTPUT
  );

  pinMode(
    PIN_DIR,
    OUTPUT
  );

  desabilitarMotor();

  digitalWrite(
    PIN_STEP,
    LOW
  );

  // ----------------------------------------------
  // I2C
  // ----------------------------------------------

  Wire.begin(
    PIN_SDA,
    PIN_SCL
  );

  // ----------------------------------------------
  // ADS1115
  // ----------------------------------------------

  if (
    !ads.begin(0x48)
  ) {

    Serial.println(
      "Erro ao iniciar ADS1115"
    );

    while (1) {
      delay(100);
    }
  }

  /*
    GAIN_FOUR:
    faixa diferencial aproximadamente
    ±1,024 V.
  */

  ads.setGain(
    GAIN_FOUR
  );

  // ----------------------------------------------
  // DS18B20
  // ----------------------------------------------

  sensorTemperatura.begin();

  // ----------------------------------------------
  // WI-FI
  // ----------------------------------------------

  WiFi.mode(
    WIFI_AP
  );

  WiFi.softAP(
    ssid,
    senha
  );

  Serial.println();

  Serial.println(
    "Rede Wi-Fi criada"
  );

  Serial.print(
    "IP: "
  );

  Serial.println(
    WiFi.softAPIP()
  );

  // ----------------------------------------------
  // SERVIDOR
  // ----------------------------------------------

  configurarServidor();

  // Cabeçalho para os dados enviados pela Serial

  Serial.println(
    "Volume_mL,"
    "Potencial_mV,"
    "Temperatura_C,"
    "Tempo_s"
  );
}

// ======================================================
// LOOP
// ======================================================

void loop() {

  server.handleClient();

  if (
    titulacaoAtiva &&
    !titulacaoPausada
  ) {

    executarCicloTitulacao();
  }

  delay(2);
}
